# Mineru Local Installation
#req: need to have docker desktop installed already
source: https://opendatalab.github.io/MinerU/quick_start/docker_deployment/

#1. create docker folder

mkdir -p docker 

#2. download mineru
wget https://gcore.jsdelivr.net/gh/opendatalab/MinerU@master/docker/global/Dockerfile


#3. turn on docker desktop

#4.  cd to docker folder, 
# note: make sure VPN is off otherwise Huggingface might deny access to models
docker build -t mineru-vllm:latest -f Dockerfile .
#other option
docker build --network=host -t mineru-vllm:latest -f Dockerfile .



docker run --gpus all `
  --shm-size 32g `
  -p 30000:30000 -p 7860:7860 -p 8000:8000 `
  --ipc=host `
  -it mineru-vllm:latest `
  /bin/bash


#service one
mineru-api --host 0.0.0.0 --port 8000

#vllm serer
mineru-vllm-server --port 30000

mineru-vllm-server --host 0.0.0.0 --port 30000


Option 2: Using docker-compose (recommended)
Download the compose.yaml file and use profiles to start specific services Opendatalab:
bashdocker compose -f compose.yaml --profile vllm-server up -d
